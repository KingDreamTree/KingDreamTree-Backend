# ML 의존성 — 버전 기준표
#
# ⚠️ 이 파일을 직접 `pip install -r` 하지 마세요.
#    torch는 환경마다 다른 인덱스에서 받아야 합니다. scripts/install_*.sh 를 쓰세요.
#
# ⚠️ 설치 스크립트 3개(install_cpu / install_local_gpu / install_runpod)는 이 파일을
#    읽지 않고 **버전을 각자 하드코딩**하고 있다. 버전을 올릴 때는 이 파일과
#    스크립트 세 곳을 같이 고칠 것 — 한 곳만 고치면 환경별로 버전이 갈라진다.
#      scripts/install_local_gpu.sh   로컬 GPU (CUDA 12.8)
#      scripts/install_cpu.sh         EC2 / CPU 전용
#      scripts/install_runpod.sh      RunPod GPU 워커
#      macOS(Apple Silicon)           README 참조 — PyPI 기본 인덱스 + SAPIENS_DEVICE=mps
#
# torch 버전 (인덱스는 스크립트가 지정)
#   CUDA 12.8 : 2.11.0+cu128   ← RTX 50 시리즈(Blackwell, sm_120)는 cu128 이상 필수
#   CPU       : 2.11.0+cpu     ← 같은 마이너 버전으로 맞춰 동작 차이를 줄인다
#   macOS     : 2.11.0         ← PyPI 기본 빌드에 MPS 가 들어있다
TORCH_VERSION_CUDA=2.11.0+cu128
TORCH_VERSION_CPU=2.11.0+cpu

# ⚠️ torchvision 은 선택이 아니다. 없으면 transformers 의 AutoImageProcessor 가
#    ImportError 로 죽어 세그멘테이션 워커가 아예 기동하지 않는다.
#    torch 와 짝이 맞는 버전이어야 한다 (torch 2.11 ↔ torchvision 0.26).
TORCHVISION_VERSION=0.26.0

# 인덱스와 무관하게 동일한 것들
#
# ⚠️ huggingface-hub은 1.x 계열이다. transformers 5.x 가 >=1.5.0,<2.0 을 요구하므로
#    0.x 를 고정하면 ResolutionImpossible 로 설치가 통째로 실패한다.
#    (버전을 올릴 때는 pip install --dry-run 으로 조합이 풀리는지 먼저 확인할 것)
transformers==5.15.0
safetensors==0.8.0
huggingface-hub==1.27.0
accelerate==1.14.0
